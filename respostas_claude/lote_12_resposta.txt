using JuMP

model = Model()

n = 249999
x_init = [i % 3 == 1 ? 3.0 : (i % 3 == 2 ? 5.0 : -3.0) for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum((x[3*i-2]-1)^2 + (x[3*i-1]-x[3*i])^2 + (x[3*i+1]-x[3*i+2])^4
        for i in 1:div(n-2,3))
)

@constraint(model, eq[k in 1:2*div(n-2,3)],
    k % 2 == 1 ?
    x[3*div(k-1,2)+1] + x[3*div(k-1,2)+2]^2 + x[3*div(k-1,2)+3] +
    x[3*div(k-1,2)+4] + x[3*div(k-1,2)+5] - 5 == 0 :
    x[3*div(k-1,2)+3]^2 - 2*(x[3*div(k-1,2)+4]+x[3*div(k-1,2)+5]) - 3 == 0
)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))
