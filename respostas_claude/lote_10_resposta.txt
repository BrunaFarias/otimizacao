### gasoil_1600.jl

using JuMP

model = Model()

# Catalytic Cracking of Gas Oil - Collocation formulation
nc = 4
ne = 2
np = 3
nm = 21
nh = 1600

tau_data = [0.0, 0.025, 0.05, 0.075, 0.10, 0.125, 0.150, 0.175, 0.20, 0.225,
            0.250, 0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.65, 0.75, 0.85, 0.95]
z_obs = [[1.0000, 0.0], [0.8105, 0.2000], [0.6208, 0.2886], [0.5258, 0.3010],
         [0.4345, 0.3215], [0.3903, 0.3123], [0.3342, 0.2716], [0.3034, 0.2551],
         [0.2735, 0.2258], [0.2405, 0.1959], [0.2283, 0.1789], [0.2071, 0.1457],
         [0.1669, 0.1198], [0.1530, 0.0909], [0.1339, 0.0719], [0.1265, 0.0561],
         [0.1200, 0.0460], [0.0990, 0.0280], [0.0870, 0.0190], [0.0770, 0.0140],
         [0.0690, 0.0100]]
rho_leg = [0.06943184420297, 0.33000947820757, 0.66999052179243, 0.93056815579703]
bc_val = [1.0, 0.0]

tf = tau_data[end]
h_coll = tf/nh
t_part = [(i-1)*h_coll for i in 1:nh+1]
fact_v = [i == 0 ? 1.0 : prod(float.(1:i)) for i in 0:nc]
itau = [min(nh, floor(Int, tau_data[i]/h_coll)+1) for i in 1:nm]

@variable(model, theta_p[1:np] >= 0.0, start=0.0)
@variable(model, v_coll[1:nh, 1:ne])
@variable(model, w_coll[1:nh, 1:nc, 1:ne])

for i in 1:itau[1], s in 1:ne; set_start_value(v_coll[i,s], bc_val[s]); end
for j in 2:nm, i in itau[j-1]+1:itau[j], s in 1:ne; set_start_value(v_coll[i,s], z_obs[j][s]); end
for i in itau[nm]+1:nh, s in 1:ne; set_start_value(v_coll[i,s], z_obs[nm][s]); end

@objective(model, Min,
    sum((sum(v_coll[itau[j],s] + sum(w_coll[itau[j],k,s]*(tau_data[j]-t_part[itau[j]])^k/(fact_v[k+1]*h_coll^(k-1)) for k in 1:nc) - z_obs[j][s] for s in 1:ne))^2 for j in 1:nm)
)

@constraint(model, ODE_IC[s in 1:ne], v_coll[1,s] == bc_val[s])
@constraint(model, continuity[i in 1:nh-1, s in 1:ne],
    v_coll[i,s] + h_coll*sum(w_coll[i,j,s]/fact_v[j+1] for j in 1:nc) == v_coll[i+1,s])

for i in 1:nh, j in 1:nc
    uc1 = v_coll[i,1] + h_coll*sum(w_coll[i,k,1]*(rho_leg[j]^k/fact_v[k+1]) for k in 1:nc)
    uc2 = v_coll[i,2] + h_coll*sum(w_coll[i,k,2]*(rho_leg[j]^k/fact_v[k+1]) for k in 1:nc)
    Duc1 = sum(w_coll[i,k,1]*(rho_leg[j]^(k-1)/fact_v[k]) for k in 1:nc)
    Duc2 = sum(w_coll[i,k,2]*(rho_leg[j]^(k-1)/fact_v[k]) for k in 1:nc)
    @constraint(model, Duc1 == -(theta_p[1]+theta_p[3])*uc1^2)
    @constraint(model, Duc2 == theta_p[1]*uc1^2 - theta_p[2]*uc2)
end

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### henon120.jl

using JuMP

model = Model()

# Henon mountain pass problem - BREAK=120
# Same structure as dirichlet models with Henon nonlinearity c(x)=|x|^(2*l)
BREAK_val = 120
NODES = 531

@variable(model, u_h[0:BREAK_val+1, 1:NODES])
@variable(model, z_h)

@objective(model, Min, z_h)
@constraint(model, lb_z, z_h >= 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))
