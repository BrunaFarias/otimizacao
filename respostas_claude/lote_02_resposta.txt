### bearing_200.jl

using JuMP

model = Model()

pi_val = 4*atan(1.0)
nx = 200
ny = 200
b_val = 10.0
e_val = 0.1
hx = 2*pi_val/(nx+1)
hy = 2*b_val/(ny+1)
wq = [(1+e_val*cos(i*hx))^3 for i in 0:nx+1]

@variable(model, v[0:nx+1, 0:ny+1] >= 0, start = 0.0)
for i in 0:nx+1
    for j in 0:ny+1
        set_start_value(v[i,j], max(sin(i*hx), 0.0))
    end
end

@objective(model, Min,
    0.5*(hx*hy/6)*sum((wq[i+1] + 2*wq[i+2])*(((v[i+1,j]-v[i,j])/hx)^2 + ((v[i,j+1]-v[i,j])/hy)^2) for i in 0:nx, j in 0:ny) +
    0.5*(hx*hy/6)*sum((2*wq[i+1] + 2*wq[i])*(((v[i-1,j]-v[i,j])/hx)^2 + ((v[i,j-1]-v[i,j])/hy)^2) for i in 1:nx+1, j in 1:ny+1) -
    hx*hy*sum(e_val*sin(i*hx)*v[i,j] for i in 0:nx+1, j in 0:ny+1)
)

@constraint(model, c1[i in 0:nx+1], v[i,0] == 0)
@constraint(model, c2[i in 0:nx+1], v[i,ny+1] == 0)
@constraint(model, c3[j in 0:ny+1], v[0,j] == 0)
@constraint(model, c4[j in 0:ny+1], v[nx+1,j] == 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### camshape_1600.jl

using JuMP

model = Model()

n = 1600
R_v = 1.0
R_min = 1.0
R_max = 2.0
alpha_val = 1.5
pi_val = 3.14159265358979
d_theta = 2*pi_val/(5*(n+1))

@variable(model, R_min <= r[1:n] <= R_max, start = (R_min+R_max)/2)

@objective(model, Max, ((pi_val*R_v)/n)*sum(r[i] for i in 1:n))

@constraint(model, convexity[i in 2:n-1],
    -r[i-1]*r[i] - r[i]*r[i+1] + 2*r[i-1]*r[i+1]*cos(d_theta) <= 0)
@constraint(model, convex_edge1,
    -R_min*r[1] - r[1]*r[2] + 2*R_min*r[2]*cos(d_theta) <= 0)
@constraint(model, convex_edge2,
    -R_min^2 - R_min*r[1] + 2*R_min*r[1]*cos(d_theta) <= 0)
@constraint(model, convex_edge3,
    -r[n-1]*r[n] - r[n]*R_max + 2*r[n-1]*R_max*cos(d_theta) <= 0)
@constraint(model, convex_edge4,
    -2*R_max*r[n] + 2*r[n]^2*cos(d_theta) <= 0)

@constraint(model, curvature[i in 1:n-1],
    -alpha_val*d_theta <= r[i+1] - r[i])
@constraint(model, curvature_edge1,
    -alpha_val*d_theta <= r[1] - R_min)
@constraint(model, curvature_edge2,
    -alpha_val*d_theta <= R_max - r[n])
@constraint(model, curvature1[i in 1:n-1],
    r[i+1] - r[i] <= alpha_val*d_theta)
@constraint(model, curvature_edge11,
    r[1] - R_min <= alpha_val*d_theta)
@constraint(model, curvature_edge21,
    R_max - r[n] <= alpha_val*d_theta)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### clnlbeam.jl

using JuMP

model = Model()

ni = 20000
alpha_val = 350.0
h = 1/ni

@variable(model, -1.0 <= t[0:ni] <= 1.0)
@variable(model, -0.05 <= x[0:ni] <= 0.05)
@variable(model, u[0:ni])

for i in 0:ni
    set_start_value(t[i], 0.05*cos(i*h))
    set_start_value(x[i], 0.05*cos(i*h))
end

@objective(model, Min,
    sum(0.5*h*(u[i+1]^2 + u[i]^2) + 0.5*alpha_val*h*(cos(t[i+1]) + cos(t[i])) for i in 0:ni-1)
)

@constraint(model, cons1[i in 0:ni-1],
    x[i+1] - x[i] - 0.5*h*(sin(t[i+1]) + sin(t[i])) == 0)
@constraint(model, cons2[i in 0:ni-1],
    t[i+1] - t[i] - 0.5*h*u[i+1] - 0.5*h*u[i] == 0)

fix(x[0], 0.0; force=true)
fix(x[ni], 0.0; force=true)
fix(t[0], 0.0; force=true)
fix(t[ni], 0.0; force=true)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))
