### optmass.jl

using JuMP

model = Model()

n_steps = 10000
speed_v = 0.01
pen_v = 0.335

@variable(model, x_v[1:2, 0:n_steps+1])
@variable(model, v_vel[1:2, 0:n_steps+1])
@variable(model, f_force[1:2, 0:n_steps])

fix(x_v[1,0], 0.0; force=true)
fix(x_v[2,0], 0.0; force=true)
fix(v_vel[1,0], speed_v; force=true)
fix(v_vel[2,0], 0.0; force=true)

@objective(model, Min,
    pen_v*(v_vel[1,n_steps+1]^2+v_vel[2,n_steps+1]^2) -
    (x_v[1,n_steps+1]^2+x_v[2,n_steps+1]^2)
)

@constraint(model, cons1[i in 1:n_steps+1, j in 1:2],
    x_v[j,i] - x_v[j,i-1] - v_vel[j,i-1]/n_steps - f_force[j,i-1]/(2*n_steps^2) == 0)
@constraint(model, cons2[i in 1:n_steps+1, j in 1:2],
    v_vel[j,i] - v_vel[j,i-1] - f_force[j,i-1]/n_steps == 0)
@constraint(model, cons3[i in 0:n_steps],
    f_force[1,i]^2 + f_force[2,i]^2 <= 1)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### qcqp1000-1c.jl

using JuMP
using LinearAlgebra
using Random

model = Model()

Random.seed!(12345)
n = 1000; ml = 100; mq = 0; pl = 50; pq = 4
sd = 1; sq = 0.01; sp = 0.01; plf = 0.2; pqf = 0.2

function gen_LQ_mat(n, sq)
    LQ = zeros(n, n)
    for i in 1:n
        LQ[i,i] = rand()
        for j in 1:i-1
            rand() < sq && (LQ[i,j] = rand()*20-10)
        end
    end
    return LQ
end

LQ = gen_LQ_mat(n, sq)
Q = sd == 1 ? LQ * LQ' : (LQ + LQ') / 2
LP = [gen_LQ_mat(n, sp) for _ in 1:mq+pq]
P = [LP[l] * LP[l]' for l in 1:mq+pq]
y_lag = randn(ml+mq)
z_lag = [i <= pl && rand() < plf ? rand()*10 : (i > pl && rand() < pqf ? rand()*10 : 0.0) for i in 1:pl+pq]
A0 = randn(ml+mq+pl+pq, n)
A = zeros(ml+mq+pl+pq, n)
for i in 1:ml+mq+pl+pq
    max_abs = maximum(abs.(A0[i,:]))
    for j in 1:n
        abs(A0[i,j]) == max_abs ? (A[i,j] = A0[i,j]) : (rand() < sq && (A[i,j] = A0[i,j]))
    end
end
xstar = randn(n)
g = -Q*xstar + A[1:ml,:]'*y_lag[1:ml]
b_eq = [sum(A[i,j]*xstar[j] for j in 1:n) + (i > ml ? 0.5*dot(xstar, P[i-ml]*xstar) : 0.0) for i in 1:ml+mq]
b_ineq = [sum(A[ml+mq+i,j]*xstar[j] for j in 1:n) + (i > pl ? 0.5*dot(xstar, P[mq+i-pl]*xstar) : 0.0) for i in 1:pl+pq]

@variable(model, x_v[1:n])
@objective(model, Min, 0.5*dot(x_v, Q*x_v) + dot(g, x_v))

for i in 1:ml
    @constraint(model, sum(A[i,j]*x_v[j] for j in 1:n) == b_eq[i])
end
for i in 1:pl
    @constraint(model, sum(A[ml+mq+i,j]*x_v[j] for j in 1:n) <= b_ineq[i])
end
for i in pl+1:pl+pq
    @constraint(model, sum(A[ml+mq+i,j]*x_v[j] for j in 1:n) + 0.5*dot(x_v, P[mq+i-pl]*x_v) <= b_ineq[i])
end

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))
