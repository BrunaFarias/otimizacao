### lane_emden40.jl

using JuMP

model = Model()

# Lane-Emden mountain pass problem - BREAK=40
BREAK_val = 40
NODES = 531

@variable(model, u_le[0:BREAK_val+1, 1:NODES])
@variable(model, z_le)

@objective(model, Min, z_le)
@constraint(model, lb_z, z_le >= 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvle10.jl

using JuMP

model = Model()

n = 250000
x_init = [i % 2 == 1 ? -1.0 : 1.0 for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum((x[2*i-1]^2)^(x[2*i]^2+1) + (x[2*i]^2)^(x[2*i-1]^2+1) for i in 1:div(n,2))
)

@constraint(model, eq[i in 1:n-2],
    (3-2*x[i+1])*x[i+1] + 1 - x[i] - 2*x[i+2] == 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvle11.jl

using JuMP

model = Model()

n = 249998
x_init = [i % 3 == 1 ? 2.0 : (i % 3 == 2 ? 1.5 : 0.5) for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum((x[3*i-2]-x[3*i-1])^2 + (x[3*i]-1)^2 + (x[3*i+1]-1)^4 + (x[3*i+2]-1)^6
        for i in 1:div(n-2,3))
)

@constraint(model, eq[k in 1:2*div(n-2,3)],
    k % 2 == 1 ?
    x[3*div(k-1,2)+1]^2*x[3*div(k-1,2)+4] + sin(x[3*div(k-1,2)+4]-x[3*div(k-1,2)+5]) - 1 <= 0 :
    x[3*div(k-1,2)+2] + x[3*div(k-1,2)+3]^4*x[3*div(k-1,2)+4]^2 - 2 <= 0
)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvle12.jl

using JuMP

model = Model()

n = 249997
x_init = [i % 4 == 1 ? 2.0 : (i % 4 == 2 ? 1.5 : (i % 4 == 3 ? -1.0 : 0.5)) for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum((x[4*i-3]-x[4*i-2])^2 + (x[4*i-2]-x[4*i-1])^2 + (x[4*i-1]-x[4*i])^4 + (x[4*i]-x[4*i+1])^4
        for i in 1:div(n-1,4))
)

@constraint(model, eq[k in 1:3*div(n-1,4)],
    k % 3 == 1 ?
    x[4*div(k-1,3)+1] + x[4*div(k-1,3)+2]^2 + x[4*div(k-1,3)+3]^3 - 3 <= 0 :
    (k % 3 == 2 ?
    x[4*div(k-1,3)+2] - x[4*div(k-1,3)+3]^2 + x[4*div(k-1,3)+4] - 1 <= 0 :
    x[4*div(k-1,3)+1]*x[4*div(k-1,3)+5] - 1 <= 0)
)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))
