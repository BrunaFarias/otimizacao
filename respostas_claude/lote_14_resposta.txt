### lukvle8.jl

using JuMP

model = Model()

n = 250000
s1 = -0.002008
s2 = -0.0019
s3 = -0.000261
h_v = 1.0/(n+1)
x_init = [i % 2 == 1 ? -1.0 : 2.0 for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum(exp(prod(x[5*i+1-j] for j in 1:5)) + 10*(
        (sum(x[5*i+1-j]^2 for j in 1:5) - 10 - s1)^2 +
        (x[5*i-3]*x[5*i-2] - 5*x[5*i-1]*x[5*i] - s2)^2 +
        (x[5*i-4]^3 + x[5*i-3]^3 + 1 - s3)^2)
        for i in 1:div(n,5))
)

@constraint(model, eq[i in 1:n-2],
    2*x[i+1] + h_v^2*(x[i+1]+h_v*(i+1)+1)^3/2 - x[i] - x[i+2] == 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvle9.jl

using JuMP

model = Model()

n = 250000
@variable(model, x[1:n], start=-1.0)

@objective(model, Min,
    sum(1e-4*(x[2*i-1]-3)^2 - (x[2*i-1]-x[2*i]) + exp(20*(x[2*i-1]-x[2*i]))
        for i in 1:div(n,2))
)

@constraint(model, eq1, 4*(x[1]-x[2]^2)+x[2]-x[3]^2+x[3]-x[4]^2 == 0)
@constraint(model, eq2, 8*x[2]*(x[2]^2-x[1])-2*(1-x[2])+4*(x[2]-x[3]^2)+x[1]^2+x[3]-x[4]^2+x[4]-x[5]^2 == 0)
@constraint(model, eq3, 8*x[3]*(x[3]^2-x[2])-2*(1-x[3])+4*(x[3]-x[4]^2)+x[2]^2-x[1]+x[4]-x[5]^2+x[1]^2+x[5]-x[6]^2 == 0)
@constraint(model, eq4, 8*x[n-2]*(x[n-2]^2-x[n-3])-2*(1-x[n-2])+4*(x[n-2]-x[n-1]^2)+x[n-3]^2-x[n-4]+x[n-1]-x[n]^2+x[n-4]^2+x[n]-x[n-5] == 0)
@constraint(model, eq5, 8*x[n-1]*(x[n-1]^2-x[n-2])-2*(1-x[n-1])+4*(x[n-1]-x[n]^2)+x[n-2]^2-x[n-3]+x[n]+x[n-3]^2-x[n-4] == 0)
@constraint(model, eq6, 8*x[n]*(x[n]^2-x[n-1])-2*(1-x[n])+x[n-1]^2-x[n-2]+x[n-2]^2-x[n-3] == 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvli10.jl

using JuMP

model = Model()

n = 250000
x_init = [i % 2 == 1 ? -1.0 : 1.0 for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum((x[2*i-1]^2)^(x[2*i]^2+1) + (x[2*i]^2)^(x[2*i-1]^2+1)
        for i in 1:div(n,2))
)

@constraint(model, ineq[i in 1:n-2],
    (3-2*x[i+1])*x[i+1] + 1 - x[i] - 2*x[i+2] <= 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvli11.jl

using JuMP

model = Model()

n = 249998
x_init = [i % 3 == 1 ? 2.0 : (i % 3 == 2 ? 1.5 : 0.5) for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum((x[3*i-2]-x[3*i-1])^2 + (x[3*i]-1)^2 + (x[3*i+1]-1)^4 + (x[3*i+2]-1)^6
        for i in 1:div(n-2,3))
)

@constraint(model, ineq[k in 1:2*div(n-2,3)],
    k % 2 == 1 ?
    x[3*div(k-1,2)+1]^2*x[3*div(k-1,2)+4] + sin(x[3*div(k-1,2)+4]-x[3*div(k-1,2)+5]) - 1 <= 0 :
    x[3*div(k-1,2)+2] + x[3*div(k-1,2)+3]^4*x[3*div(k-1,2)+4]^2 - 2 <= 0
)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvli12.jl

using JuMP

model = Model()

n = 249997
x_init = [i % 4 == 1 ? 2.0 : (i % 4 == 2 ? 1.5 : (i % 4 == 3 ? -1.0 : 0.5)) for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum((x[4*i-3]-x[4*i-2])^2 + (x[4*i-2]-x[4*i-1])^2 + (x[4*i-1]-x[4*i])^4 + (x[4*i]-x[4*i+1])^4
        for i in 1:div(n-1,4))
)

@constraint(model, ineq[k in 1:3*div(n-1,4)],
    k % 3 == 1 ?
    x[4*div(k-1,3)+1] + x[4*div(k-1,3)+2]^2 + x[4*div(k-1,3)+3]^3 - 3 <= 0 :
    (k % 3 == 2 ?
    x[4*div(k-1,3)+2] - x[4*div(k-1,3)+3]^2 + x[4*div(k-1,3)+4] - 1 <= 0 :
    x[4*div(k-1,3)+1]*x[4*div(k-1,3)+5] - 1 <= 0)
)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))
