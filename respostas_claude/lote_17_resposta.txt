### lukvli6.jl

using JuMP

model = Model()

n = 249999
@variable(model, x[1:n+1], start=3.0)

@objective(model, Min,
    sum(abs((2+5*x[i]^2)*x[i] + 1 + sum(x[j]*(1+x[j]) for j in max(i-5,1):min(i+1,n)))^(7/3)
        for i in 1:n)
)

@constraint(model, ineq[i in 1:div(n,2)],
    4*x[2*i] - (x[2*i-1]-x[2*i+1])*exp(x[2*i-1]-x[2*i]-x[2*i+1]) - 3 >= 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvli7.jl

using JuMP

model = Model()

n = 250000
x_init = [i >= 1 && i <= n ? 1.0 : 0.0 for i in 0:n+1]

@variable(model, x[0:n+1])
for i in 0:n+1; set_start_value(x[i], x_init[i+1]); end

@objective(model, Min,
    sum(i*(1-cos(x[i])+sin(x[i-1])-sin(x[i+1])) for i in 1:n)
)

@constraint(model, ineq1, 4*(x[1]-x[2]^2)+x[2]-x[3]^2 <= 0)
@constraint(model, ineq2, 8*x[2]*(x[2]^2-x[1])-2*(1-x[2])+4*(x[2]-x[3]^2)+x[3]-x[4]^2 <= 0)
@constraint(model, ineq3, 8*x[n-1]*(x[n-1]^2-x[n-2])-2*(1-x[n])+4*(x[n-1]-x[n]^2)+x[n-2]^2-x[n-3] <= 0)
@constraint(model, ineq4, 8*x[n]*(x[n]^2-x[n-1])-2*(1-x[n])+x[n-1]^2-x[n-2] <= 0)

fix(x[0], 0.0; force=true)
fix(x[n+1], 0.0; force=true)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

### lukvli8.jl

using JuMP

model = Model()

n = 250000
s1 = -0.002008
s2 = -0.0019
s3 = -0.000261
h_v = 1.0/(n+1)
x_init = [i % 2 == 1 ? -1.0 : 2.0 for i in 1:n]

@variable(model, x[1:n])
for i in 1:n; set_start_value(x[i], x_init[i]); end

@objective(model, Min,
    sum(exp(prod(x[5*i+1-j] for j in 1:5)) + 10*(
        (sum(x[5*i+1-j]^2 for j in 1:5) - 10 - s1)^2 +
        (x[5*i-3]*x[5*i-2] - 5*x[5*i-1]*x[5*i] - s2)^2 +
        (x[5*i-4]^3 + x[5*i-3]^3 + 1 - s3)^2)
        for i in 1:div(n,5))
)

@constraint(model, ineq[i in 1:n-2],
    2*x[i+1] + h_v^2*(x[i+1]+h_v*(i+1)+1)^3/2 - x[i] - x[i+2] <= 0)

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))

@variable(model, g_rate[1:ne-1] >= 0.0)
@variable(model, m_rate[1:ne] >= 0.0)
@variable(model, v_coll[1:nh, 1:ne])
@variable(model, w_coll[1:nh, 1:nc, 1:ne])

for i in 1:itau[1], s in 1:ne; set_start_value(v_coll[i,s], z_obs[1][s]); end
for j in 2:nm, i in itau[j-1]+1:itau[j], s in 1:ne; set_start_value(v_coll[i,s], z_obs[j][s]); end

    sum((sum(v_coll[itau[j],s] + sum(w_coll[itau[j],k,s]*(tau_data[j]-t_part[itau[j]])^k/(fact_v[k+1]*h_coll^(k-1)) for k in 1:nc) - z_obs[j][s] for s in 1:ne))^2 for j in 1:nm)
)

    v_coll[i,s] + h_coll*sum(w_coll[i,j,s]/fact_v[j+1] for j in 1:nc) == v_coll[i+1,s])

for i in 1:nh, j in 1:nc
    uc = [v_coll[i,s] + h_coll*sum(w_coll[i,k,s]*(rho_leg[j]^k/fact_v[k+1]) for k in 1:nc) for s in 1:ne]
    Duc = [sum(w_coll[i,k,s]*(rho_leg[j]^(k-1)/fact_v[k]) for k in 1:nc) for s in 1:ne]
    @constraint(model, Duc[1] == -(m_rate[1]+g_rate[1])*uc[1])
    for s in 2:ne-1
        @constraint(model, Duc[s] == g_rate[s-1]*uc[s-1] - (m_rate[s]+g_rate[s])*uc[s])
    end
    @constraint(model, Duc[ne] == g_rate[ne-1]*uc[ne-1] - m_rate[ne]*uc[ne])
end

optimize!(model)
println("Status: ", termination_status(model))
println("Objetivo: ", objective_value(model))
